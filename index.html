🛠️ Developer Integration Playground

LAYER 4 PROTOTYPE

SolAudit provides a composable, zero-friction integration experience. Use Anchor macros to secure any smart contract instruction with a single decorator, or integrate via our TypeScript SDK in client apps.

use anchor_lang::prelude::*;
use solaudit_sdk::{compliance_gate, ComplianceGated, PredicateType};

#[program]
pub mod solana_defi_protocol {
    use super::*;

    // Automatically gates trade function with a KYC check
    #[compliance_gate(predicate = PredicateType::KycAml, issuer_set = "solaudit_mainnet")]
    pub fn execute_swap(ctx: Context<ExecuteSwap>, amount: u64) -> Result<()> {
        // Your trade, lending, or liquidity provisioning business logic
        msg!("Compliance verification check passed atomically via PDA verify_attestation CPI call.");
        Ok(())
    }
}
import { Connection, PublicKey } from '@solana/web3.js';
import { SolAuditClient, PredicateType } from '@solaudit/sdk';

const connection = new Connection("https://api.mainnet-beta.solana.com");
const client = new SolAuditClient({ connection });

// Query the user's active attestation credentials
const walletAddress = new PublicKey("3tGhNkPqBmVrLwDsAeCjXfYuIoZpMnQvKsWlRdEaTbFc");
const attestation = await client.getAttestation(walletAddress, PredicateType.KYC_AML_COMBINED);

console.log("Attestation Address:", attestation.publicKey.toBase58());
console.log("Issuer Details:", attestation.issuer.name);
console.log("Is Verification Active?:", attestation.status === "ACTIVE" && attestation.expiresAt > Date.now());
// Construct Solana Pay payment requests requiring KYC attestation prior to execution
import { encodeURL } from '@solana/pay';

const paymentParameters = {
    recipient: new PublicKey("7xK9mQ3f..."),
    amount: new BigNumber(125.50),
    label: "Superteam Nomad Store",
    message: "Purchase validation with AML clearance attestation",
    // Embed the verification standard required to clear transaction
    memo: "solaudit:KYC_AML_COMBINED:v1.0.0"
};

const solanaPayURL = encodeURL(paymentParameters);
console.log("Compliant checkout QR code URI:", solanaPayURL);
